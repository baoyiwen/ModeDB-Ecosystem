import fs from 'node:fs';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {spawnSync} from 'node:child_process';
const base='docs/security-v1-deliveries/evidence/SEC-M7-D20/';
const sha=b=>createHash('sha256').update(b).digest('hex');
const descriptor=p=>({path:p,sha256:sha(fs.readFileSync(p))});
const receipt=JSON.parse(fs.readFileSync(base+'acceptance.json','utf8'));
const verification=JSON.parse(fs.readFileSync(base+'verification.json','utf8'));
const checks=JSON.parse(fs.readFileSync(base+'checks.original.json.txt','utf8'));
for(const file of receipt.historicalEvidence)assert.equal(sha(fs.readFileSync(file.path)),file.sha256,'HISTORY_CHANGED: '+file.path);
const code=fs.readFileSync(base+'documents-check.original.mjs.txt','utf8');
const result=spawnSync(process.execPath,['--input-type=commonjs','-e',code],{encoding:'utf8',windowsHide:true});
fs.writeFileSync(base+'documents.original.json.txt',result.stdout);fs.writeFileSync(base+'documents.stderr.original.txt',result.stderr);assert.equal(result.status,0,result.stdout+result.stderr);
verification.documentValidation=JSON.parse(result.stdout);
checks.documentCheck={script:base+'documents-check.original.mjs.txt',exitCode:0,log:'documents.original.json.txt'};
checks.historicalEvidenceVerified=receipt.historicalEvidence.length;
fs.writeFileSync(base+'checks.original.json.txt',JSON.stringify(checks,null,2)+'\n');
fs.writeFileSync(base+'verification.json',JSON.stringify(verification,null,2)+'\n');
receipt.verification.status='passed';
receipt.files=receipt.files.map(file=>({...descriptor(file.path),sha256Lf:sha(fs.readFileSync(file.path,'utf8').replace(/\r\n/g,'\n'))}));
receipt.verificationArtifacts=fs.readdirSync(base).filter(n=>n!=='acceptance.json').sort().map(n=>descriptor(base+n));
fs.writeFileSync(base+'acceptance.json',JSON.stringify(receipt,null,2)+'\n');
console.log(JSON.stringify({ok:true,developmentFiles:receipt.files.length,historical:receipt.historicalEvidence.length,artifacts:receipt.verificationArtifacts.length,documents:verification.documentValidation},null,2));
